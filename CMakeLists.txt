cmake_minimum_required(VERSION 3.22)
project(TheCuelist VERSION 0.5.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/JUCE/CMakeLists.txt")
  message(FATAL_ERROR "JUCE assente. Eseguire scripts/build.ps1 per scaricare la versione fissata.")
endif()
add_subdirectory(vendor/JUCE EXCLUDE_FROM_ALL)
add_library(CuelistCore STATIC src/ShowModel.cpp src/AudioEngine.cpp src/MidiClip.cpp src/MidiOutputRouter.cpp src/OscClip.cpp src/OscOutputRouter.cpp)
target_include_directories(CuelistCore PUBLIC src vendor/JUCE/modules)
target_compile_definitions(CuelistCore PUBLIC JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_MODAL_LOOPS_PERMITTED=0 JUCE_ASIO=0 JUCE_USE_MP3AUDIOFORMAT=1)
target_link_libraries(CuelistCore PRIVATE juce::juce_audio_utils PUBLIC juce::juce_recommended_config_flags)
if(MSVC)
  target_compile_options(CuelistCore PUBLIC /utf-8 /W4)
endif()
enable_testing()
add_executable(CuelistTests tests/Tests.cpp src/MainComponent.cpp src/CueViews.cpp src/Timeline.cpp src/ColumnUtilities.cpp src/PianoRoll.cpp src/MidiUI.cpp)
target_link_libraries(CuelistTests PRIVATE CuelistCore)
add_test(NAME CuelistTests COMMAND CuelistTests)
juce_add_gui_app(TheCuelist PRODUCT_NAME "The Cuelist" COMPANY_NAME "The Cuelist" BUNDLE_ID "app.thecuelist.desktop")
target_sources(TheCuelist PRIVATE src/Main.cpp src/MainComponent.cpp src/CueViews.cpp src/Timeline.cpp src/ColumnUtilities.cpp src/PianoRoll.cpp src/MidiUI.cpp)
target_link_libraries(TheCuelist PRIVATE CuelistCore)

cmake_minimum_required(VERSION 3.22)
project(TheCuelist VERSION 0.9.0 LANGUAGES C CXX)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION.txt" THE_CUELIST_DISPLAY_VERSION LIMIT_COUNT 1)
if(NOT THE_CUELIST_DISPLAY_VERSION MATCHES "^0\\.9\\.0(-[a-zA-Z0-9.]+)?$")
  message(FATAL_ERROR "VERSION.txt must match the release's CMake version.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/JUCE/CMakeLists.txt")
  message(FATAL_ERROR "JUCE missing. Run scripts/build.ps1 to download the pinned version.")
endif()
add_subdirectory(vendor/JUCE EXCLUDE_FROM_ALL)
juce_add_binary_data(CuelistAssets SOURCES assets/thecuelist_icon.png)
add_library(CuelistCore STATIC src/MidiSessionPolicy.cpp src/ShowModel.cpp src/ControlDispatchPolicy.cpp src/ControlLayerBuilder.cpp src/ShowEditOperations.cpp src/ProjectSafety.cpp src/ProjectMediaLoader.cpp src/AudioEngine.cpp src/MidiClip.cpp src/MidiOutputRouter.cpp src/OscClip.cpp src/OscOutputRouter.cpp src/ControlOutputRouter.cpp)
target_include_directories(CuelistCore PUBLIC src vendor/JUCE/modules)
target_compile_definitions(CuelistCore PUBLIC JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_MODAL_LOOPS_PERMITTED=0 JUCE_ASIO=0 JUCE_USE_MP3AUDIOFORMAT=1)
target_link_libraries(CuelistCore PRIVATE juce::juce_audio_utils PUBLIC juce::juce_recommended_config_flags)
if(MSVC)
  target_compile_options(CuelistCore PUBLIC /utf-8 /W4)
endif()
enable_testing()
add_executable(CuelistTests tests/Tests.cpp src/ControlUI.cpp src/ControlRuntime.cpp src/MainComponent.cpp src/InspectorController.cpp src/ProjectLifecycle.cpp src/CueOperations.cpp src/CellOperations.cpp src/WorkspaceLayout.cpp src/CueViews.cpp src/Timeline.cpp src/ColumnUtilities.cpp src/GainMeterFader.cpp src/PianoRoll.cpp src/MidiUI.cpp src/OscEditor.cpp src/OscUI.cpp)
target_link_libraries(CuelistTests PRIVATE CuelistCore CuelistAssets)
add_test(NAME CuelistTests COMMAND CuelistTests)
# Optional operator-side hardware inventory and OSC 127.0.0.1 self-test.
# Never run physical I/O tests automatically on cloud CI hardware.
add_executable(CuelistHardwareProbe tests/HardwareProbe.cpp)
target_link_libraries(CuelistHardwareProbe PRIVATE CuelistCore)

juce_add_gui_app(TheCuelist PRODUCT_NAME "The Cuelist" COMPANY_NAME "The Cuelist" BUNDLE_ID "app.thecuelist.desktop" ICON_BIG "${CMAKE_CURRENT_SOURCE_DIR}/assets/thecuelist_icon.png")
target_sources(TheCuelist PRIVATE src/ControlUI.cpp src/ControlRuntime.cpp src/Main.cpp src/MainComponent.cpp src/InspectorController.cpp src/ProjectLifecycle.cpp src/CueOperations.cpp src/CellOperations.cpp src/WorkspaceLayout.cpp src/CueViews.cpp src/Timeline.cpp src/ColumnUtilities.cpp src/GainMeterFader.cpp src/PianoRoll.cpp src/MidiUI.cpp src/OscEditor.cpp src/OscUI.cpp)
target_link_libraries(TheCuelist PRIVATE CuelistCore CuelistAssets)

# Product version and distributable name are derived from VERSION.txt.
target_compile_definitions(TheCuelist PRIVATE THE_CUELIST_VERSION=\"${THE_CUELIST_DISPLAY_VERSION}\")
